# lease 微服务 数据库建表语句

> 版本：v1.0 ｜ 日期：2026-08-08
> 说明：本文件为**规划阶段交付物**，尚未执行。按"每服务独立库"分组，共 28 张表。
> 通用约定（对应源码 `BaseEntity`）：
> - `id` bigint 主键自增
> - `create_time` / `update_time` 由 MyBatis-Plus 的 `MybatisMetaObjectHandler` 自动填充
> - `is_deleted` 逻辑删除，默认 0（`@TableLogic`）
> 所有表：`ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci`

---

## 一、新建数据库

```sql
CREATE DATABASE IF NOT EXISTS `lease_auth`      DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;
CREATE DATABASE IF NOT EXISTS `lease_apartment` DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;
CREATE DATABASE IF NOT EXISTS `lease_agreement` DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;
CREATE DATABASE IF NOT EXISTS `lease_user`      DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;
```

## 二、lease_auth（auth-service，认证服务）

> 当前无业务表。认证服务只依赖 Redis（验证码/邮箱验证码缓存）+ 通过 Feign 调用 user 服务获取用户数据。
> 后续如需登录日志等表再在此库追加。

### 3.1 apartment_info（公寓信息表）

```sql
DROP TABLE IF EXISTS `apartment_info`;
CREATE TABLE `apartment_info` (
  `id`            bigint        NOT NULL AUTO_INCREMENT COMMENT '主键',
  `name`          varchar(50)   DEFAULT NULL COMMENT '公寓名称',
  `introduction`  varchar(255)  DEFAULT NULL COMMENT '公寓介绍',
  `district_id`   bigint        DEFAULT NULL COMMENT '所处区域id',
  `district_name` varchar(50)   DEFAULT NULL COMMENT '所处区域名称',
  `city_id`       bigint        DEFAULT NULL COMMENT '所处城市id',
  `city_name`     varchar(50)   DEFAULT NULL COMMENT '所处城市名称',
  `province_id`   bigint        DEFAULT NULL COMMENT '所处省份id',
  `province_name` varchar(50)   DEFAULT NULL COMMENT '所处省份名称',
  `address_detail` varchar(255) DEFAULT NULL COMMENT '详细地址',
  `latitude`      varchar(20)   DEFAULT NULL COMMENT '经度',
  `longitude`     varchar(20)   DEFAULT NULL COMMENT '纬度',
  `phone`         varchar(20)   DEFAULT NULL COMMENT '公寓前台电话',
  `is_release`    tinyint(1)    NOT NULL DEFAULT 0 COMMENT '是否发布：1已发布 0未发布',
  `create_time`   datetime      DEFAULT NULL COMMENT '创建时间',
  `update_time`   datetime      DEFAULT NULL COMMENT '更新时间',
  `is_deleted`    tinyint(1)    NOT NULL DEFAULT 0 COMMENT '逻辑删除',
  PRIMARY KEY (`id`),
  KEY `idx_district_id` (`district_id`),
  KEY `idx_city_id` (`city_id`),
  KEY `idx_province_id` (`province_id`)
) ENGINE=InnoDB AUTO_INCREMENT=1 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci COMMENT='公寓信息表';
```

### 4.1 lease_agreement（租约信息表）

```sql
DROP TABLE IF EXISTS `lease_agreement`;
CREATE TABLE `lease_agreement` (
  `id`                  bigint        NOT NULL AUTO_INCREMENT COMMENT '主键',
  `phone`               varchar(20)   DEFAULT NULL COMMENT '承租人手机号码',
  `name`                varchar(50)   DEFAULT NULL COMMENT '承租人姓名',
  `identification_number` varchar(18) DEFAULT NULL COMMENT '承租人身份证号码',
  `apartment_id`        bigint        DEFAULT NULL COMMENT '签约公寓id',
  `room_id`             bigint        DEFAULT NULL COMMENT '签约房间id',
  `lease_start_date`    date          DEFAULT NULL COMMENT '租约开始日期',
  `lease_end_date`      date          DEFAULT NULL COMMENT '租约结束日期',
  `lease_term_id`       bigint        DEFAULT NULL COMMENT '租期id',
  `rent`                decimal(10,2) DEFAULT NULL COMMENT '租金（元/月）',
  `deposit`             decimal(10,2) DEFAULT NULL COMMENT '押金（元）',
  `payment_type_id`     bigint        DEFAULT NULL COMMENT '支付类型id',
  `status`              tinyint(1)    NOT NULL DEFAULT 1 COMMENT '租约状态：1签约待确认 2已签约 3已取消 4已到期 5退租待确认 6已退租 7续约待确认',
  `source_type`         tinyint(1)    NOT NULL DEFAULT 1 COMMENT '租约来源：1新签 2续约',
  `additional_info`     varchar(255)  DEFAULT NULL COMMENT '备注信息',
  `create_time`         datetime      DEFAULT NULL COMMENT '创建时间',
  `update_time`         datetime      DEFAULT NULL COMMENT '更新时间',
  `is_deleted`          tinyint(1)    NOT NULL DEFAULT 0 COMMENT '逻辑删除',
  PRIMARY KEY (`id`),
  KEY `idx_apartment_id` (`apartment_id`),
  KEY `idx_room_id` (`room_id`),
  KEY `idx_lease_term_id` (`lease_term_id`),
```
